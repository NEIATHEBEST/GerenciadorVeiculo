import React,{useState, useEffect} from 'react';
import { StyleSheet, Text, View, Modal, TouchableOpacity, SafeAreaView, TextInput, Alert, ScrollView, Platform } from 'react-native';
import axios from 'axios'; // npm i axios


/*Para instalar Animatable=> npm install react-native-animatable --save*/
import * as Animatable from 'react-native-animatable';
import {Ionicons} from '@expo/vector-icons';




export default function App(){

   function mensagemDelete(id){

     // WEB
  if (Platform.OS === "web") {
    const confirmar = window.confirm(
      "Deseja excluir este Registro?"
    );

    if (confirmar) {
      deleteItem(id);
    }

    return;
  }
    
     Alert.alert(
      "Excluir Registro",
      "Deseja Excluir este Registro?",
       [
         {
           text: "Não",
           onPress: () => console.log("Cancel Pressed"),
           style: "cancel"
         },
         { text: "Sim", onPress: () => deleteItem(id) }
       ],
       { cancelable: true }
     );  
  }

  
  const mensagemDuplicidade = () =>
    Alert.alert(
      "Erro ao Salvar",
      "atributo2 Já Cadastrado",
      [
        
        { text: "OK", onPress: () => setAbrir(true) }
      ],
      { cancelable: true }
    ); 

  const mensagemDadosIncorretos = () =>
    Alert.alert(
      "Erro ao Logar",
      "Dados Incorretos",
      [
        {text: "OK" , onPress: () => setAbrirLogin(true)}
      ],
      {cancelable: true}
    )

async function login(){
  const obj = {atributo2,atributo3};
  const res = await axios.post(api + 'login.php', obj);
  if(res.data.success == 'Dados Incorretos!'){
    mensagemDadosIncorretos();
  }
  else{
    setAbrirLogin(false);
  }
}


  const api = 'http://localhost//';
  const [lista, setLista] = useState([]);
  const [atributo1, setatributo1] = useState('');
  const [atributo2, setatributo2] = useState('');
  const [atributo3, setatributo3] = useState('');
  const [id, setId] = useState('');
  const [buscar, setBuscar] = useState('');
  {/* state para tela de cadastro modal */}
  const[abrir, setAbrir] = useState(false);

  {/*state para login */}
  const[abrirLogin,setAbrirLogin] = useState(false);

  useEffect(()=> {
    listarDados();
  },[])



    
  async function listarDados(){
    const res = await axios.get(api + 'listar.php?busca=' + buscar);
   if(res.data.success === true){
         setLista(res.data.result);
       }

    //console.log(res.data.result);    
  }

    function buscarDados(){
        listarDados();
    }

  async function add(){
    const obj = {atributo1, atributo2, atributo3, id};    
    //npm i axios  - PARA INSTALAR O AXIOS NO EXPO
    if(id > 0){
      const res = await axios.post(api + 'editar.php', obj);
       if(res.data.success === true){
         //mensagemSalvar;
         limparCampos();
       }
     }else{
     const res = await axios.post(api + 'add.php', obj);
     
      if(res.data.success === true){
        //mensagemSalvar();
        limparCampos();        
      }

      if(res.data.success === 'atributo2 já Cadastrado!'){
        mensagemDuplicidade();
        
      }     
    }
    
    listarDados();
    setAbrir(false);   
  }

   async function getItem(id){
     const res = await axios.get(api + 'buscarId.php?id=' + id);
     setId(res.data.id);
     setatributo1(res.data.atributo1);
     setatributo2(res.data.atributo2);
     setatributo3(res.data.atributo3);
     setAbrir(true);   
    
    }

   async function deleteItem(id){
     const res = await axios.get(api + 'excluir.php?id=' + id);
     listarDados();
   }


  function limparCampos(){
        setatributo1('');
        setatributo2('');
        setatributo3('');
        setId('0');
  }


  return (
    <View>
    <View style={estilos.navbar}>
      
    <Text style={estilos.textonavbar}>Lista de Usuários</Text>      
    
    <TouchableOpacity
      
      style={estilos.botao}
      onPress={ () => setAbrir(true)}
      >
        <Ionicons name="add-outline" size={30} color="#FFF"></Ionicons>
        
      </TouchableOpacity>
    </View>

    <View style={estilos.ViewinputBuscar}>
    <TextInput 
      style={estilos.inputBuscar}
      placeholder="Buscar pelo atributo1"
      value={buscar}
      onChangeText={ (buscar) => setBuscar(buscar)}
      onChange={buscarDados()}
      />      
     <Ionicons style={estilos.iconeBuscar} name="search-outline" size={25} color="#4b4a49"></Ionicons>
     
    </View>
   
    <ScrollView>
    <View style={estilos.grid}>
    
      {lista.map(item => (
        <View style={estilos.griditem} key={item.id}><Text style={{color: '#585858'}}>{item.id} - {item.atributo1} - {item.atributo2}</Text>

        <TouchableOpacity style={estilos.gridbotaoEditar}
             onPress={() => getItem(item.id)} >
               <Ionicons name="create-outline" size={30} color="#50b9e1"></Ionicons>
        </TouchableOpacity> 
        
        <TouchableOpacity style={estilos.gridbotaoExcluir}
            onPress={() => mensagemDelete(item.id)} >
              <Ionicons name="trash-outline" size={30} color="#e15f50"></Ionicons>
        </TouchableOpacity>                                  
      </View>
     ))}        
    </View>
    </ScrollView>

   
{/*tela login */ }      
 <Modal
      animationType="slide"
      transparent={false}
      visible={abrirLogin}
      >
 <SafeAreaView style={estilos.modal}>         

    <View style={estilos.modalHeader}>

       <Text style={estilos.textoModal}>Faça seu Login</Text>
    </View>

<Animatable.View  
animation="bounceInUp"
useNativeDriver  >

<TextInput 
 type="atributo2"
 style={estilos.input}
 placeholder="Insira seu atributo2"
 value={atributo2}
 onChangeText={ (atributo2) => setatributo2(atributo2)}
/>
<TextInput 
 secureTextEntry={true}
 style={estilos.input}
 placeholder="Insira sua atributo3"
 value={atributo3}
 onChangeText={ (atributo3) => setatributo3(atributo3)}
/>

<TouchableOpacity  
      style={estilos.botaoModal}
      onPress={login}
      >
    <Text  style={estilos.textoBotaoModal}>Login</Text>
</TouchableOpacity>
    </Animatable.View>
   </SafeAreaView>
  </Modal>    


{/*tela cadastro usuario */ }
     <Modal
      animationType="slide"
      transparent={false}
      visible={abrir}
      >
        <SafeAreaView style={estilos.modal}>

          <View style={estilos.modalHeader}>
          <TouchableOpacity
              onPress={ () => setAbrir(false)}
          >
           <Ionicons style={{marginLeft:5, marginRight:5}} name="arrow-back-outline" size={35} color="#FFF"></Ionicons>
          </TouchableOpacity>
        <Text style={estilos.textoModal}>Inserir Usuário</Text>
        </View>

{/*Animação cadastro usuario */ }
    <Animatable.View  
        animation="bounceInUp"
        useNativeDriver  >


   <TextInput 
      type="text"
      style={estilos.input}
      placeholder="Insira um atributo1"
      value={atributo1}
      onChangeText={ (atributo1) => setatributo1(atributo1)}
      />

      <TextInput 
      style={estilos.input}
      placeholder="Insira seu atributo2"
      value={atributo2}
      onChangeText={ (atributo2) => setatributo2(atributo2)}
      />

<TextInput 
      style={estilos.input}
      placeholder="Insira sua atributo3"
      value={atributo3}
      onChangeText={ (atributo3) => setatributo3(atributo3)}
      secureTextEntry={true}
      />

      
      <TouchableOpacity  
      style={estilos.botaoModal}
      onPress={add}
      >
        <Text  style={estilos.textoBotaoModal}>Salvar</Text>
      </TouchableOpacity>

  

    </Animatable.View>

        </SafeAreaView>
 
      </Modal>


 </View> 
    );
}
